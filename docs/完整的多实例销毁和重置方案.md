  1. 创建状态重置函数                                                                                                             
  /**
   * 重置所有关键的鼠标操作状态
   * 当销毁多实例面板时调用此函数，确保后续鼠标操作不被错误处理
   */
  function resetMouseOperationStates() {
    console.log('[状态重置] 开始重置鼠标操作状态');

    // 获取 SyncManager 实例
    const syncManager = window.__syncManager__;
    if (!syncManager) {
      console.warn('[状态重置] SyncManager 不存在');
      return;
    }

    // 1. 重置操作状态（operationState）
    if (syncManager.operationState) {
      syncManager.operationState = {
        isDragging: false,
        operationType: null,  // 关键：重置当前操作类型
        lastMousePos: { x: 0, y: 0 },
        operationStartTime: 0
      };
      console.log('[状态重置] ✓ operationState 已重置');
    }

    // 2. 重置鼠标状态（mouseState）
    if (syncManager.mouseState) {
      syncManager.mouseState = {
        isDown: false,           // 关键：重置鼠标按下状态
        button: -1,
        lastX: 0,
        lastY: 0,
        dragStartX: 0,
        dragStartY: 0
      };
      console.log('[状态重置] ✓ mouseState 已重置');
    }

    // 3. 清理翻转检测状态
    if (typeof syncManager._flipDetectionDone !== 'undefined') {
      syncManager._flipDetectionDone = false;
      console.log('[状态重置] ✓ _flipDetectionDone 已重置');
    }

    // 4. 重置滚轮操作状态
    if (typeof syncManager.isWheeling !== 'undefined') {
      syncManager.isWheeling = false;
      console.log('[状态重置] ✓ isWheeling 已重置');
    }

    // 5. 重置当前操作类型
    if (typeof syncManager.currentOperation !== 'undefined') {
      syncManager.currentOperation = null;
      console.log('[状态重置] ✓ currentOperation 已重置');
    }

    // 6. 重置统一坐标系初始化标志
    if (typeof syncManager.unifiedProjectionInitialized !== 'undefined') {
      syncManager.unifiedProjectionInitialized = false;
      console.log('[状态重置] ✓ unifiedProjectionInitialized 已重置');
    }

    // 7. 重置局部坐标系使用标志
    if (typeof syncManager.useLocalCoordSystem !== 'undefined') {
      syncManager.useLocalCoordSystem = false;
      console.log('[状态重置] ✓ useLocalCoordSystem 已重置');
    }

    // 8. 重置同步深度计数器（防止循环同步）
    if (typeof syncManager.syncDepth !== 'undefined') {
      syncManager.syncDepth = 0;
      console.log('[状态重置] ✓ syncDepth 已重置');
    }

    // 9. 清除节流定时器
    if (syncManager.throttleTimer) {
      clearTimeout(syncManager.throttleTimer);
      syncManager.throttleTimer = null;
      console.log('[状态重置] ✓ throttleTimer 已清除');
    }

    console.log('[状态重置] ✓ 所有鼠标操作状态重置完成');
  }

  /**
   * 恢复 Cesium 输入并清理所有相关状态
   */
  function restoreCesiumAndResetStates() {
    console.log('[多实例销毁] 开始恢复 Cesium 并重置状态');

    // 1. 恢复 Cesium 输入
    const cesiumViewer = window.__cesiumViewer__ || window.viewer;
    if (cesiumViewer && cesiumViewer.scene) {
      cesiumViewer.scene.screenSpaceCameraController.enableInputs = true;
      console.log('[多实例销毁] ✓ Cesium 输入已恢复');
    }

    // 2. 重置所有鼠标操作状态
    resetMouseOperationStates();

    // 3. 重新初始化 SyncManager 和 Cesium 引用
    requestAnimationFrame(() => {
      ensureSyncManagerInitialized();

      // 额外：触发 Cesium 渲染刷新
      if (cesiumViewer && cesiumViewer.scene) {
        cesiumViewer.scene.requestRender();
        console.log('[多实例销毁] ✓ Cesium 渲染已刷新');
      }
    });

    console.log('[多实例销毁] ✓ 所有状态重置完成');
  }

    if (multiInstanceContainer) {
      multiInstanceContainer.remove();
      console.log('[多实例销毁] ✓ 多实例容器已移除');
    }

    // 2. 恢复单例显示
    const singletonContainer = document.getElementById('singleton-container') ||
                               document.querySelector('#app') ||
                               document.querySelector('[data-singleton]');
    if (singletonContainer) {
      singletonContainer.classList.remove('hidden');
      console.log('[多实例销毁] ✓ 单例显示已恢复');
    }

    // 3. ⭐ 关键修复：恢复 Cesium 输入并重置所有状态
    restoreCesiumAndResetStates();

    console.log('[多实例销毁] ========== 销毁流程完成 ==========');
  }
